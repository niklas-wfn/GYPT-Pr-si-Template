% --- Dynamische Segmentleiste ---
\newcommand{\numsecs}{1}
\makeatletter
\AtEndDocument{%
  \immediate\write\@auxout{\string\gdef\string\numsecs{\the\value{section}}}%
}
\makeatother

\newcommand{\secbar}{%
  \begin{tikzpicture}[x=\paperwidth, y=1pt, inner sep=0pt, outer sep=0pt]
    \foreach \i in {1,...,\numsecs}{
      \ifnum\value{section}>\i\relax
        \colorlet{segcol}{gyptg}%
      \else\ifnum\value{section}=\i\relax
        \colorlet{segcol}{gyptb}%
      \else
        \colorlet{segcol}{gyptgrey}%
      \fi\fi
      \fill[segcol] ({(\i-1+0.06)/\numsecs},0) rectangle ({(\i-0.06)/\numsecs},4);
    }
  \end{tikzpicture}%
}
